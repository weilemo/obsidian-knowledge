背景

实验室设备资源有限，预约流程如果只依赖人工登记或前端判断，容易出现时间冲突、资质不清、维护占用未同步和违约难追踪等问题。

时间冲突同一设备可能被多个用户预约在重叠时间段。
资质审核高风险设备需要确认用户是否具备有效资质。
维护占用设备维护期间应自动阻止预约。
违约追踪no_show 等行为应自动留下记录。

SQL 语句

1. 表结构：预约表
CREATE TABLE reservations (
  reservation_id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id        INTEGER NOT NULL,
  device_id      INTEGER NOT NULL,
  start_time     TEXT NOT NULL,
  end_time       TEXT NOT NULL,
  status         TEXT NOT NULL DEFAULT 'pending'
    CHECK (status IN ('pending','approved','cancelled','completed','no_show')),
  purpose        TEXT,
  FOREIGN KEY (user_id) REFERENCES users(user_id),
  FOREIGN KEY (device_id) REFERENCES devices(device_id)
);
2. 视图：未来预约详情
CREATE VIEW vw_upcoming_reservations AS
SELECT r.reservation_id, u.name, d.device_name, l.lab_name,
       r.start_time, r.end_time, r.status, r.purpose
FROM reservations r
JOIN users u ON r.user_id = u.user_id
JOIN devices d ON r.device_id = d.device_id
JOIN labs l ON d.lab_id = l.lab_id
WHERE r.end_time > datetime('now')
ORDER BY r.start_time;
3. 触发器：阻止同设备时间冲突
CREATE TRIGGER trg_reservation_overlap_ins
BEFORE INSERT ON reservations
WHEN NEW.status IN ('pending', 'approved')
BEGIN
  SELECT CASE WHEN EXISTS (
    SELECT 1 FROM reservations r
    WHERE r.device_id = NEW.device_id
      AND r.status IN ('pending', 'approved')
      AND NOT (NEW.end_time <= r.start_time
               OR NEW.start_time >= r.end_time)
  )
  THEN RAISE(ABORT, '预约失败：该时间段设备已被预约')
  END;
END;
